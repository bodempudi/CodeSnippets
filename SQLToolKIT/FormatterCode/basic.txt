using Microsoft.SqlServer.TransactSql.ScriptDom;
using NotepadPlusPlusPlugin.Formatting.Diagnostics;
using NotepadPlusPlusPlugin.Formatting.Emission;
using NBK.SQL.CodingStandards.Core.Parse;
using System;
using System.Collections.Generic;
using NBK.SQL.CodingStandards.Core.Models;

namespace NotepadPlusPlusPlugin.Formatting.Validation
{
    public sealed class FormatterIdempotencyResult
    {
        public bool Passed
        {
            get;
            set;
        }

        public string SecondPassSql
        {
            get;
            set;
        }

        public string Message
        {
            get;
            set;
        }

        /*
         * Diagnostics for the first difference between
         * formatting pass 1 and formatting pass 2.
         */

        public int FirstDifferenceIndex
        {
            get;
            set;
        }

        public int FirstDifferenceLine
        {
            get;
            set;
        }

        public int FirstDifferenceColumn
        {
            get;
            set;
        }

        public string FirstPassLine
        {
            get;
            set;
        }

        public string SecondPassLine
        {
            get;
            set;
        }
    }

    public sealed class FormatterIdempotencyValidator
    {
        private readonly SqlParserService _parserService;

        public FormatterIdempotencyValidator(
            SqlParserService parserService)
        {
            if (parserService == null)
            {
                throw new ArgumentNullException(
                    "parserService");
            }

            _parserService =
                parserService;
        }

        public FormatterIdempotencyResult Validate(
            string formattedSql)
        {
            string firstPass =
                formattedSql ??
                string.Empty;

            SqlParseResult parseResult =
                _parserService.Parse(
                    firstPass);

            if (!parseResult.IsValid)
            {
                return new FormatterIdempotencyResult
                {
                    Passed =
                        false,

                    SecondPassSql =
                        firstPass,

                    FirstDifferenceIndex =
                        -1,

                    FirstDifferenceLine =
                        0,

                    FirstDifferenceColumn =
                        0,

                    FirstPassLine =
                        string.Empty,

                    SecondPassLine =
                        string.Empty,

                    Message =
                        "Idempotency check could not parse " +
                        "the first formatted output."
                };
            }

            IList<TSqlParserToken> tokens =
                parseResult.Tokens ??
                new List<TSqlParserToken>();

            LayoutPlan plan =
                new LayoutPlan();

            GenericAstWalker walker =
                new GenericAstWalker();

            FormatterDiagnostics diagnostics =
                new FormatterDiagnostics();

            LayoutPolicyEngine engine =
                new LayoutPolicyEngine(
                    tokens,
                    walker,
                    plan,
                    diagnostics);

            engine.Analyze(
                parseResult.Fragment);

            TokenPreservingEmitter emitter =
                new TokenPreservingEmitter();

            string secondPass =
                emitter.Emit(
                    tokens,
                    plan);

            bool passed =
                string.Equals(
                    firstPass,
                    secondPass,
                    StringComparison.Ordinal);

            if (passed)
            {
                return new FormatterIdempotencyResult
                {
                    Passed =
                        true,

                    SecondPassSql =
                        secondPass,

                    FirstDifferenceIndex =
                        -1,

                    FirstDifferenceLine =
                        0,

                    FirstDifferenceColumn =
                        0,

                    FirstPassLine =
                        string.Empty,

                    SecondPassLine =
                        string.Empty,

                    Message =
                        "Second formatting pass is identical " +
                        "to the first output."
                };
            }

            /*
             * ============================================================
             * FIND FIRST DIFFERENCE
             * ============================================================
             */

            int differenceIndex =
                FindFirstDifference(
                    firstPass,
                    secondPass);

            int line;
            int column;

            GetLineAndColumn(
                firstPass,
                differenceIndex,
                out line,
                out column);

            string firstPassLine =
                GetLineText(
                    firstPass,
                    line);

            string secondPassLine =
                GetLineText(
                    secondPass,
                    line);

            string message =
                BuildFailureMessage(
                    differenceIndex,
                    line,
                    column,
                    firstPassLine,
                    secondPassLine);

            return new FormatterIdempotencyResult
            {
                Passed =
                    false,

                SecondPassSql =
                    secondPass,

                FirstDifferenceIndex =
                    differenceIndex,

                FirstDifferenceLine =
                    line,

                FirstDifferenceColumn =
                    column,

                FirstPassLine =
                    firstPassLine,

                SecondPassLine =
                    secondPassLine,

                Message =
                    message
            };
        }

        /*
         * ================================================================
         * FIRST DIFFERENCE
         * ================================================================
         */

        private static int FindFirstDifference(
            string first,
            string second)
        {
            string left =
                first ??
                string.Empty;

            string right =
                second ??
                string.Empty;

            int commonLength =
                Math.Min(
                    left.Length,
                    right.Length);

            for (int index = 0;
                 index < commonLength;
                 index++)
            {
                if (left[index] != right[index])
                {
                    return index;
                }
            }

            /*
             * Every character in the common portion matched,
             * but one string is longer than the other.
             */

            if (left.Length != right.Length)
            {
                return commonLength;
            }

            return -1;
        }

        /*
         * ================================================================
         * LINE / COLUMN
         * ================================================================
         */

        private static void GetLineAndColumn(
            string text,
            int index,
            out int line,
            out int column)
        {
            string value =
                text ??
                string.Empty;

            if (index < 0)
            {
                line =
                    0;

                column =
                    0;

                return;
            }

            if (index > value.Length)
            {
                index =
                    value.Length;
            }

            line =
                1;

            column =
                1;

            for (int i = 0;
                 i < index;
                 i++)
            {
                char current =
                    value[i];

                if (current == '\r')
                {
                    /*
                     * Treat CRLF as one newline.
                     */

                    if (i + 1 < index &&
                        value[i + 1] == '\n')
                    {
                        i++;
                    }

                    line++;

                    column =
                        1;

                    continue;
                }

                if (current == '\n')
                {
                    line++;

                    column =
                        1;

                    continue;
                }

                column++;
            }
        }

        /*
         * ================================================================
         * LINE TEXT
         * ================================================================
         */

        private static string GetLineText(
            string text,
            int targetLine)
        {
            if (string.IsNullOrEmpty(text) ||
                targetLine <= 0)
            {
                return string.Empty;
            }

            int currentLine =
                1;

            int lineStart =
                0;

            for (int i = 0;
                 i <= text.Length;
                 i++)
            {
                bool endOfText =
                    i == text.Length;

                bool lineBreak =
                    !endOfText &&
                    (text[i] == '\r' ||
                     text[i] == '\n');

                if (!endOfText &&
                    !lineBreak)
                {
                    continue;
                }

                if (currentLine == targetLine)
                {
                    return text.Substring(
                        lineStart,
                        i - lineStart);
                }

                if (endOfText)
                {
                    break;
                }

                if (text[i] == '\r' &&
                    i + 1 < text.Length &&
                    text[i + 1] == '\n')
                {
                    i++;
                }

                currentLine++;

                lineStart =
                    i + 1;
            }

            return string.Empty;
        }

        /*
         * ================================================================
         * FAILURE MESSAGE
         * ================================================================
         */

        private static string BuildFailureMessage(
            int differenceIndex,
            int line,
            int column,
            string firstPassLine,
            string secondPassLine)
        {
            return
                "Second formatting pass differs from the first output. " +
                "First difference at Line=" +
                line +
                ", Column=" +
                column +
                ", CharacterIndex=" +
                differenceIndex +
                ". FirstPass=[" +
                (firstPassLine ?? string.Empty) +
                "]. SecondPass=[" +
                (secondPassLine ?? string.Empty) +
                "].";
        }
    }
}
