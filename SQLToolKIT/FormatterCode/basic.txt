private static bool IsGeneratedSectionSeparator(
    TSqlParserToken token)
{
    if (token == null ||
        token.TokenType != TSqlTokenType.SingleLineComment)
    {
        return false;
    }

    string text =
        (token.Text ?? string.Empty)
            .TrimEnd(
                ' ',
                '\t',
                '\r',
                '\n');

    if (text.Length < 2)
    {
        return false;
    }

    for (int i = 0;
         i < text.Length;
         i++)
    {
        if (text[i] != '-')
        {
            return false;
        }
    }

    int startColumn =
        token.Column;

    if (startColumn <= 0)
    {
        return false;
    }

    int endColumn =
        (startColumn - 1) +
        text.Length;

    return
        endColumn == 135;
}
