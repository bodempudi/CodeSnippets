public string Emit(
    IList<TSqlParserToken> tokens,
    LayoutPlan plan)
{
    if (tokens == null)
    {
        throw new ArgumentNullException(
            "tokens");
    }

    if (plan == null)
    {
        throw new ArgumentNullException(
            "plan");
    }

    Dictionary<int, LayoutInstruction> instructions =
        plan.Instructions
            .GroupBy(
                x => x.TokenIndex)
            .ToDictionary(
                x => x.Key,
                x => x.Last());

    StringBuilder output =
        new StringBuilder();

    for (int i = 0;
         i < tokens.Count;
         i++)
    {
        TSqlParserToken token =
            tokens[i];

        LayoutInstruction instruction;

        /*
         * ============================================================
         * APPLY EXPLICIT LAYOUT FIRST
         * ============================================================
         */
        if (instructions.TryGetValue(
                i,
                out instruction))
        {
            ApplyInstruction(
                output,
                instruction);
        }

        /*
         * ============================================================
         * EOF
         * ============================================================
         */
        if (token.TokenType ==
            TSqlTokenType.EndOfFile)
        {
            continue;
        }

        /*
         * ============================================================
         * WHITESPACE
         * ============================================================
         */
        if (token.TokenType ==
            TSqlTokenType.WhiteSpace)
        {
            int next =
                FindNextNonWhitespaceToken(
                    tokens,
                    i + 1);

            /*
             * The upcoming token owns its whitespace.
             *
             * Example:
             *
             * old whitespace
             * SELECT
             *
             * If SELECT has a NewLine instruction,
             * discard the old whitespace completely.
             */
            if (next >= 0 &&
                instructions.ContainsKey(
                    next))
            {
                continue;
            }

            /*
             * ========================================================
             * EXISTING PHYSICAL NEWLINE
             * ========================================================
             */
            if (ContainsNewline(
                    token.Text))
            {
                int previous =
                    FindPreviousNonWhitespaceToken(
                        tokens,
                        i - 1);

                TSqlParserToken previousToken =
                    previous >= 0
                        ? tokens[previous]
                        : null;

                TSqlParserToken nextToken =
                    next >= 0
                        ? tokens[next]
                        : null;

                /*
                 * ====================================================
                 * COMMENT NEWLINE PROTECTION
                 * ====================================================
                 *
                 * Developer comment boundaries are source-owned.
                 *
                 * Never collapse an existing physical newline when
                 * either side is a comment.
                 *
                 * Examples:
                 *
                 * AS
                 * <multiline history comment>
                 *
                 * END
                 * ----------------------------------------------------
                 *
                 * -- Developer comment
                 * SELECT ...
                 *
                 * A line containing only hyphens is parsed by
                 * ScriptDom as a SingleLineComment, so it is
                 * protected by this rule as well.
                 */
                bool previousIsComment =
                    IsCommentToken(
                        previousToken);

                bool nextIsComment =
                    IsCommentToken(
                        nextToken);

                if (previousIsComment ||
                    nextIsComment)
                {
                    output.Append(
                        token.Text);

                    continue;
                }

                /*
                 * Normally preserve an original newline.
                 *
                 * However, when the PREVIOUS token was explicitly
                 * positioned by the layout engine, this old source
                 * newline must not undo that formatter decision.
                 *
                 * Example:
                 *
                 * FROM
                 *     dbo.Customer
                 *
                 * where FROM already received its layout position.
                 *
                 * We want:
                 *
                 * FROM dbo.Customer
                 */
                if (previous >= 0 &&
                    instructions.ContainsKey(
                        previous))
                {
                    if (ShouldInsertSpace(
                            previousToken,
                            nextToken))
                    {
                        RemoveTrailingWhitespace(
                            output);

                        if (output.Length > 0)
                        {
                            output.Append(
                                ' ');
                        }
                    }
                    else
                    {
                        RemoveTrailingWhitespace(
                            output);
                    }

                    continue;
                }

                /*
                 * No formatter instruction owns this boundary.
                 * Preserve the developer/source newline.
                 */
                output.Append(
                    token.Text);

                continue;
            }

            /*
             * ========================================================
             * HORIZONTAL WHITESPACE ONLY
             * ========================================================
             */
            int previousSameLine =
                FindPreviousNonWhitespaceToken(
                    tokens,
                    i - 1);

            if (previousSameLine < 0 ||
                next < 0)
            {
                continue;
            }

            TSqlParserToken previousTokenSameLine =
                tokens[previousSameLine];

            TSqlParserToken nextTokenSameLine =
                tokens[next];

            if (ShouldInsertSpace(
                    previousTokenSameLine,
                    nextTokenSameLine))
            {
                /*
                 * Multiple spaces/tabs become one space.
                 */
                RemoveTrailingWhitespace(
                    output);

                if (output.Length > 0)
                {
                    output.Append(
                        ' ');
                }
            }
            else
            {
                /*
                 * Punctuation-sensitive boundary.
                 *
                 * Examples:
                 *
                 * dbo.Customer
                 * COUNT(*)
                 * a,
                 */
                RemoveTrailingWhitespace(
                    output);
            }

            continue;
        }

        /*
         * ============================================================
         * TOKEN TEXT
         * ============================================================
         *
         * Never reconstruct or modify SQL token text.
         *
         * This protects:
         *
         * - string literals
         * - comments
         * - identifiers
         * - quoted identifiers
         * - numbers
         * - operators
         */
        output.Append(
            token.Text);
    }

    return
        output.ToString();
}
