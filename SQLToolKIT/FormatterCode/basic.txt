using Microsoft.SqlServer.TransactSql.ScriptDom;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace NotepadPlusPlusPlugin.Formatting.Emission
{
    public sealed class TokenPreservingEmitter
    {
        private const string IndentText =
            "    ";

        private const int SectionSeparatorEndColumn =
            135;

        public string Emit(
            IList<TSqlParserToken> tokens,
            LayoutPlan plan)
        {
            if (tokens == null)
            {
                throw new ArgumentNullException(
                    "tokens");
            }

            if (plan == null)
            {
                throw new ArgumentNullException(
                    "plan");
            }

            Dictionary<int, LayoutInstruction> instructions =
                plan.Instructions
                    .GroupBy(
                        x => x.TokenIndex)
                    .ToDictionary(
                        x => x.Key,
                        x => x.Last());

            StringBuilder output =
                new StringBuilder();

            for (int i = 0;
                 i < tokens.Count;
                 i++)
            {
                TSqlParserToken token =
                    tokens[i];

                LayoutInstruction instruction;

                /*
                 * ==================================================
                 * APPLY EXPLICIT LAYOUT FIRST
                 * ==================================================
                 */

                if (instructions.TryGetValue(
                        i,
                        out instruction))
                {
                    ApplyInstruction(
                        output,
                        instruction);
                }

                /*
                 * ==================================================
                 * EOF
                 * ==================================================
                 */

                if (token.TokenType ==
                    TSqlTokenType.EndOfFile)
                {
                    continue;
                }

                /*
                 * ==================================================
                 * WHITESPACE
                 * ==================================================
                 */

                if (token.TokenType ==
                    TSqlTokenType.WhiteSpace)
                {
                    int next =
                        FindNextNonWhitespaceToken(
                            tokens,
                            i + 1);

                    /*
                     * The upcoming token owns its whitespace.
                     *
                     * Example:
                     *
                     * old whitespace
                     * SELECT
                     *
                     * If SELECT has a NewLine instruction,
                     * discard the old whitespace completely.
                     */
                    if (next >= 0 &&
                        instructions.ContainsKey(
                            next))
                    {
                        continue;
                    }

                    /*
                     * ------------------------------------------------
                     * EXISTING PHYSICAL NEWLINE
                     * ------------------------------------------------
                     *
                     * Normally preserve an original newline.
                     *
                     * However, when the PREVIOUS token was explicitly
                     * positioned by the layout engine, the old source
                     * newline must not undo that formatter decision.
                     *
                     * Example:
                     *
                     * FROM
                     *     dbo.Customer
                     *
                     * where FROM already received its layout position.
                     *
                     * We want:
                     *
                     * FROM dbo.Customer
                     */
                    if (ContainsNewLine(
                            token.Text))
                    {
                        int previous =
                            FindPreviousNonWhitespaceToken(
                                tokens,
                                i - 1);

                        if (previous >= 0 &&
                            instructions.ContainsKey(
                                previous))
                        {
                            TSqlParserToken previousToken =
                                tokens[previous];

                            TSqlParserToken nextToken =
                                next >= 0
                                    ? tokens[next]
                                    : null;

                            if (ShouldInsertSpace(
                                    previousToken,
                                    nextToken))
                            {
                                RemoveTrailingWhitespace(
                                    output);

                                if (output.Length > 0)
                                {
                                    output.Append(
                                        ' ');
                                }
                            }
                            else
                            {
                                RemoveTrailingWhitespace(
                                    output);
                            }

                            continue;
                        }

                        /*
                         * No formatter instruction owns this boundary.
                         * Preserve the developer/source newline.
                         */
                        output.Append(
                            token.Text);

                        continue;
                    }

                    /*
                     * ------------------------------------------------
                     * HORIZONTAL WHITESPACE ONLY
                     * ------------------------------------------------
                     */

                    int previousSameLine =
                        FindPreviousNonWhitespaceToken(
                            tokens,
                            i - 1);

                    if (previousSameLine < 0 ||
                        next < 0)
                    {
                        continue;
                    }

                    TSqlParserToken previousTokenSameLine =
                        tokens[previousSameLine];

                    TSqlParserToken nextTokenSameLine =
                        tokens[next];

                    if (ShouldInsertSpace(
                            previousTokenSameLine,
                            nextTokenSameLine))
                    {
                        /*
                         * Multiple spaces/tabs become one space.
                         */
                        RemoveTrailingWhitespace(
                            output);

                        if (output.Length > 0)
                        {
                            output.Append(
                                ' ');
                        }
                    }
                    else
                    {
                        /*
                         * Punctuation-sensitive pair:
                         *
                         * dbo . Customer
                         * COUNT ( *
                         * a ,
                         *
                         * becomes:
                         *
                         * dbo.Customer
                         * COUNT(*
                         * a,
                         */
                        RemoveTrailingWhitespace(
                            output);
                    }

                    continue;
                }

                /*
                 * ==================================================
                 * TOKEN TEXT
                 * ==================================================
                 *
                 * Never reconstruct or modify SQL token text.
                 *
                 * This protects:
                 *
                 * - string literals
                 * - comments
                 * - identifiers
                 * - quoted identifiers
                 * - numbers
                 * - operators
                 */

                output.Append(
                    token.Text);
            }

            return
                output.ToString();
        }

        /*
         * ==========================================================
         * APPLY LAYOUT ACTION
         * ==========================================================
         */

        private static void ApplyInstruction(
            StringBuilder output,
            LayoutInstruction instruction)
        {
            if (instruction == null)
            {
                return;
            }

            if (instruction.Action ==
                LayoutAction.Tight)
            {
                ApplyTight(
                    output);

                return;
            }

            if (instruction.Action ==
                LayoutAction.Space)
            {
                ApplySpace(
                    output);

                return;
            }

            if (instruction.Action ==
                LayoutAction.BlankLine)
            {
                ApplyBlankLine(
                    output,
                    instruction.IndentLevel);

                return;
            }

            if (instruction.Action ==
                LayoutAction.SectionSeparator)
            {
                ApplySectionSeparator(
                    output,
                    instruction.IndentLevel);

                return;
            }

            if (instruction.Action ==
                LayoutAction.NewLine)
            {
                ApplyNewLine(
                    output,
                    instruction.IndentLevel);
            }
        }

        /*
         * ==========================================================
         * NEW LINE
         * ==========================================================
         */

        private static void ApplyNewLine(
            StringBuilder output,
            int indentLevel)
        {
            RemoveTrailingWhitespace(
                output);

            if (!EndsWithNewLine(
                    output))
            {
                output.AppendLine();
            }

            output.Append(
                RepeatIndent(
                    indentLevel));
        }

        /*
         * ==========================================================
         * BLANK LINE
         * ==========================================================
         */

        private static void ApplyBlankLine(
            StringBuilder output,
            int indentLevel)
        {
            RemoveTrailingWhitespaceIncludingNewLines(
                output);

            if (output.Length > 0)
            {
                output.AppendLine();
                output.AppendLine();
            }

            output.Append(
                RepeatIndent(
                    indentLevel));
        }

        /*
         * ==========================================================
         * SECTION SEPARATOR
         * ==========================================================
         *
         * Generated separator ends at absolute column 135.
         */

        private static void ApplySectionSeparator(
            StringBuilder output,
            int indentLevel)
        {
            RemoveTrailingWhitespaceIncludingNewLines(
                output);

            /*
             * Avoid duplicate adjacent formatter-generated
             * separators when two structural boundaries meet.
             */
            RemoveTrailingGeneratedSectionSeparator(
                output);

            RemoveTrailingWhitespaceIncludingNewLines(
                output);

            if (output.Length > 0)
            {
                output.AppendLine();
            }

            string indent =
                RepeatIndent(
                    indentLevel);

            output.Append(
                indent);

            int hyphenCount =
                SectionSeparatorEndColumn -
                indent.Length;

            /*
             * Must remain a valid SQL single-line comment.
             */
            if (hyphenCount < 2)
            {
                hyphenCount = 2;
            }

            output.Append(
                new string(
                    '-',
                    hyphenCount));

            output.AppendLine();

            output.Append(
                indent);
        }

        /*
         * ==========================================================
         * REMOVE TRAILING GENERATED SECTION SEPARATOR
         * ==========================================================
         */

        private static void RemoveTrailingGeneratedSectionSeparator(
            StringBuilder output)
        {
            if (output == null ||
                output.Length == 0)
            {
                return;
            }

            int lineStart =
                output.Length - 1;

            while (lineStart >= 0 &&
                   output[lineStart] != '\r' &&
                   output[lineStart] != '\n')
            {
                lineStart--;
            }

            lineStart++;

            string lastLine =
                output.ToString(
                    lineStart,
                    output.Length - lineStart);

            if (!IsGeneratedSectionSeparatorLine(
                    lastLine))
            {
                return;
            }

            output.Length =
                lineStart;
        }

        /*
         * ==========================================================
         * GENERATED SECTION SEPARATOR LINE
         * ==========================================================
         */

        private static bool IsGeneratedSectionSeparatorLine(
            string line)
        {
            if (string.IsNullOrEmpty(
                    line))
            {
                return false;
            }

            string normalized =
                line.TrimEnd(
                    ' ',
                    '\t',
                    '\r',
                    '\n');

            /*
             * Full physical width must end at column 135.
             */
            if (normalized.Length !=
                SectionSeparatorEndColumn)
            {
                return false;
            }

            int index =
                0;

            /*
             * Formatter indentation consists of spaces.
             */
            while (index <
                       normalized.Length &&
                   normalized[index] == ' ')
            {
                index++;
            }

            int hyphenCount =
                normalized.Length -
                index;

            if (hyphenCount < 2)
            {
                return false;
            }

            for (int i = index;
                 i < normalized.Length;
                 i++)
            {
                if (normalized[i] != '-')
                {
                    return false;
                }
            }

            return true;
        }

        /*
         * ==========================================================
         * TIGHT
         * ==========================================================
         */

        private static void ApplyTight(
            StringBuilder output)
        {
            RemoveTrailingWhitespaceIncludingNewLines(
                output);
        }

        /*
         * ==========================================================
         * SPACE
         * ==========================================================
         */

        private static void ApplySpace(
            StringBuilder output)
        {
            RemoveTrailingWhitespaceIncludingNewLines(
                output);

            if (output.Length > 0)
            {
                output.Append(
                    ' ');
            }
        }

        /*
         * ==========================================================
         * SAME-LINE SPACING DECISION
         * ==========================================================
         *
         * Called only when an actual whitespace boundary existed.
         *
         * Result:
         *
         * true  -> normalize to exactly one space
         * false -> normalize to no space
         */

        private static bool ShouldInsertSpace(
            TSqlParserToken previous,
            TSqlParserToken current)
        {
            if (previous == null ||
                current == null)
            {
                return false;
            }

            string previousText =
                previous.Text ??
                string.Empty;

            string currentText =
                current.Text ??
                string.Empty;

            /*
             * ------------------------------------------------------
             * NO SPACE BEFORE THESE
             * ------------------------------------------------------
             *
             * dbo.Customer
             * a,b
             * COUNT(*)
             * END;
             */

            if (currentText == "." ||
                currentText == "," ||
                currentText == ")" ||
                currentText == ";")
            {
                return false;
            }

            /*
             * ------------------------------------------------------
             * NO SPACE AFTER THESE
             * ------------------------------------------------------
             *
             * dbo.Customer
             * (CustomerID
             */

            if (previousText == "." ||
                previousText == "(")
            {
                return false;
            }

            /*
             * ------------------------------------------------------
             * NORMAL TOKEN BOUNDARY
             * ------------------------------------------------------
             *
             * Existing horizontal whitespace becomes one space.
             */

            return true;
        }

        /*
         * ==========================================================
         * DOES WHITESPACE CONTAIN A PHYSICAL NEWLINE?
         * ==========================================================
         */

        private static bool ContainsNewLine(
            string text)
        {
            if (string.IsNullOrEmpty(
                    text))
            {
                return false;
            }

            return
                text.IndexOf('\r') >= 0 ||
                text.IndexOf('\n') >= 0;
        }

        /*
         * ==========================================================
         * NEXT NON-WHITESPACE TOKEN
         * ==========================================================
         */

        private static int FindNextNonWhitespaceToken(
            IList<TSqlParserToken> tokens,
            int start)
        {
            for (int i = start;
                 i < tokens.Count;
                 i++)
            {
                if (tokens[i].TokenType !=
                    TSqlTokenType.WhiteSpace)
                {
                    return i;
                }
            }

            return -1;
        }

        /*
         * ==========================================================
         * PREVIOUS NON-WHITESPACE TOKEN
         * ==========================================================
         */

        private static int FindPreviousNonWhitespaceToken(
            IList<TSqlParserToken> tokens,
            int start)
        {
            for (int i = start;
                 i >= 0;
                 i--)
            {
                if (tokens[i].TokenType !=
                    TSqlTokenType.WhiteSpace)
                {
                    return i;
                }
            }

            return -1;
        }

        /*
         * ==========================================================
         * INDENT
         * ==========================================================
         */

        private static string RepeatIndent(
            int indentLevel)
        {
            if (indentLevel <= 0)
            {
                return
                    string.Empty;
            }

            return
                string.Concat(
                    Enumerable.Repeat(
                        IndentText,
                        indentLevel));
        }

        /*
         * ==========================================================
         * REMOVE HORIZONTAL WHITESPACE
         * ==========================================================
         */

        private static void RemoveTrailingWhitespace(
            StringBuilder builder)
        {
            while (builder.Length > 0)
            {
                char value =
                    builder[
                        builder.Length - 1];

                if (value != ' ' &&
                    value != '\t')
                {
                    break;
                }

                builder.Length--;
            }
        }

        /*
         * ==========================================================
         * REMOVE ALL TRAILING WHITESPACE
         * ==========================================================
         */

        private static void RemoveTrailingWhitespaceIncludingNewLines(
            StringBuilder builder)
        {
            while (builder.Length > 0)
            {
                char value =
                    builder[
                        builder.Length - 1];

                if (value != ' ' &&
                    value != '\t' &&
                    value != '\r' &&
                    value != '\n')
                {
                    break;
                }

                builder.Length--;
            }
        }

        /*
         * ==========================================================
         * ENDS WITH NEW LINE
         * ==========================================================
         */

        private static bool EndsWithNewLine(
            StringBuilder builder)
        {
            if (builder.Length == 0)
            {
                return false;
            }

            char last =
                builder[
                    builder.Length - 1];

            return
                last == '\r' ||
                last == '\n';
        }
    }
}
