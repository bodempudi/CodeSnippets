private static void FormatYourSQL()
{
    try
    {
        string sql =
            GetCurrentEditorText();

        if (string.IsNullOrWhiteSpace(sql))
        {
            return;
        }

        var formatterService =
            new SqlFormatterService();

        SqlFormatterResult formatResult =
            formatterService.Format(sql);

        var formatResults =
            new List<SqlResultInfo>();

        if (formatResult == null)
        {
            formatResults.Add(
                new SqlResultInfo
                {
                    Source = "Format",
                    Severity = "Error",
                    Line = 0,
                    Column = 0,
                    Message =
                        "SQL formatter did not return a result."
                });

            EnsureErrorPanel();

            _errorPanel.SetTitle(
                "SQL Format Results");

            _errorPanel.ShowResults(
                formatResults);

            ShowErrorPanel();

            return;
        }

        if (!formatResult.ValidationPassed)
        {
            formatResults.Add(
                new SqlResultInfo
                {
                    Source = "Format",
                    Severity = "Error",
                    Line = 0,
                    Column = 0,
                    Message =
                        string.IsNullOrWhiteSpace(
                            formatResult.Message)
                            ? "SQL formatting could not be completed safely. Original SQL was preserved."
                            : formatResult.Message
                });

            EnsureErrorPanel();

            _errorPanel.SetTitle(
                "SQL Format Results");

            _errorPanel.ShowResults(
                formatResults);

            ShowErrorPanel();

            return;
        }

        if (formatResult.WasFormatted &&
            !string.IsNullOrEmpty(
                formatResult.Sql))
        {
            var editor =
                new ScintillaGateway(
                    PluginBase.GetCurrentScintilla());

            editor.SetText(
                formatResult.Sql);

            formatResults.Add(
                new SqlResultInfo
                {
                    Source = "Format",
                    Severity = "Information",
                    Line = 0,
                    Column = 0,
                    Message =
                        string.IsNullOrWhiteSpace(
                            formatResult.Message)
                            ? "SQL formatted successfully."
                            : "SQL formatted successfully. " +
                              formatResult.Message
                });
        }
        else
        {
            formatResults.Add(
                new SqlResultInfo
                {
                    Source = "Format",
                    Severity = "Information",
                    Line = 0,
                    Column = 0,
                    Message =
                        "SQL is already formatted. No changes were required."
                });
        }

        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Format Results");

        _errorPanel.ShowResults(
            formatResults);

        ShowErrorPanel();
    }
    catch (Exception ex)
    {
        var formatResults =
            new List<SqlResultInfo>();

        formatResults.Add(
            new SqlResultInfo
            {
                Source = "Format",
                Severity = "Error",
                Line = 0,
                Column = 0,
                Message = ex.ToString()
            });

        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Format Results");

        _errorPanel.ShowResults(
            formatResults);

        ShowErrorPanel();
    }
}
