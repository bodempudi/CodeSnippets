private static void AddElse(
    ScalarExpression elseExpression,
    int caseIndent,
    AstNodeContext context,
    LayoutContext layoutContext)
{
    if (elseExpression == null)
    {
        return;
    }

    int elseIndex =
        FindPreviousToken(
            layoutContext,
            elseExpression.FirstTokenIndex,
            TSqlTokenType.Else);

    if (elseIndex < 0)
    {
        return;
    }

    /*
     * ELSE = CASE + 1
     *
     * Keep the ELSE expression on the same line.
     *
     * Example:
     *
     * CASE
     *     WHEN Condition = 1
     *         THEN Value
     *     ELSE NULL
     * END
     */

    layoutContext.AddInstruction(
        elseIndex,
        caseIndent + 1,
        "CaseLayoutPolicy",
        context);

    /*
     * Tighten the expression following ELSE so that
     * simple values such as NULL, 0, '', column names,
     * variables, etc. remain on the ELSE line.
     */

    layoutContext.AddTightenInstruction(
        elseExpression.FirstTokenIndex,
        "CaseLayoutPolicy",
        context);
}
