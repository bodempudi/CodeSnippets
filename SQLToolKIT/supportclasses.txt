private static void FormatSql()
{
    try
    {
        string sql =
            PluginBase.GetCurrentEditorText();

        if (string.IsNullOrWhiteSpace(sql))
        {
            MessageBox.Show(
                "No SQL text found.",
                "SQL Formatter",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information);

            return;
        }

        SqlFormatterOptions options =
            new SqlFormatterOptions
            {
                EnableLogFile = false,
                EnablePdfReport = false
            };

        SqlFormatterService formatter =
            new SqlFormatterService(
                options);

        SqlFormatterResult result =
            formatter.Format(
                sql);

        if (result == null)
        {
            MessageBox.Show(
                "Formatter returned no result.",
                "SQL Formatter",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);

            return;
        }

        if (!result.Success)
        {
            MessageBox.Show(
                result.Message,
                "SQL Formatter",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);

            return;
        }

        PluginBase.ReplaceCurrentEditorText(
            result.FormattedSql);
    }
    catch (Exception ex)
    {
        MessageBox.Show(
            ex.Message,
            "SQL Formatter",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
    }
}
