using Microsoft.SqlServer.TransactSql.ScriptDom;
using NBK.SQL.CodingStandards.Core.Models;
using System.Collections.Generic;
using System.IO;
using System.Linq;

namespace NBK.SQL.CodingStandards.Core.Parsing
{
    public sealed class SQLParserService
    {
        public SqlParseResult ParseSql(
            string sql)
        {
            var parser =
                new TSql160Parser(false);

            TSqlFragment fragment;

            IList<ParseError> errors;

            using (var reader =
                new StringReader(sql))
            {
                fragment =
                    parser.Parse(
                        reader,
                        out errors);
            }

            return new SqlParseResult
            {
                Fragment =
                    fragment,

                Errors =
                    errors
                        .Select(
                            error =>
                                new SqlResultInfo
                                {
                                    Line =
                                        error.Line,

                                    Column =
                                        error.Column,

                                    Message =
                                        error.Message
                                })
                        .ToList()
            };
        }
    }
}
