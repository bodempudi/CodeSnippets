private static void ReviewYourSQL()
{
    string sql =
        GetCurrentEditorText();

    var parserService =
        new SQLParserService();

    SqlParseResult parseResult =
        parserService.ParseSql(
            sql);

    if (!parseResult.IsValid)
    {
        _errorPanel.ShowResults(
            parseResult.Errors);

        return;
    }

    var reviewContext =
        new SqlReviewContext
        {
            Sql =
                sql,

            Fragment =
                parseResult.Fragment
        };

    var reviewService =
        new SqlReviewService();

    IEnumerable<ReviewFinding> findings =
        reviewService.Review(
            reviewContext);

    // Show findings
}
