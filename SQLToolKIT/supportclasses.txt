private static void ParseSQL()
{
    try
    {
        string sql =
            GetCurrentEditorText();

        IList<ParseError> errors =
            ParseSql(sql);

        var results =
            new List<SqlResultInfo>();

        if (errors != null &&
            errors.Count > 0)
        {
            foreach (
                ParseError error
                in errors)
            {
                results.Add(
                    new SqlResultInfo
                    {
                        Source = "Parser",
                        Severity = "Error",
                        Line = error.Line,
                        Column = error.Column,
                        Message = error.Message
                    });
            }
        }
        else
        {
            results.Add(
                new SqlResultInfo
                {
                    Source = "Parser",
                    Severity = "Info",
                    Line = 0,
                    Column = 0,
                    Message = "No SQL syntax errors found."
                });
        }

        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Parsing Results");

        _errorPanel.ShowResults(
            results);

        ShowErrorPanel();
    }
    catch (Exception ex)
    {
        MessageBox.Show(
            ex.ToString(),
            "Parse Your SQL",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
    }
}
