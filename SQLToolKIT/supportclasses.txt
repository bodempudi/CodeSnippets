private static void ParseYourSQL()
{
    try
    {
        string sql =
            GetCurrentEditorText();

        var parserService =
            new SQLParserService();

        var parseResult =
            parserService.ParseSql(sql);

        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Parsing Results");

        if (!parseResult.IsValid)
        {
            _errorPanel.ShowResults(
                parseResult.Errors);

            ShowErrorPanel();

            return;
        }

        var results =
            new List<SqlResultInfo>();

        results.Add(
            new SqlResultInfo
            {
                Source = "Parse",
                Severity = "Info",
                Line = 0,
                Column = 0,
                Message = "No SQL parsing errors found."
            });

        _errorPanel.ShowResults(
            results);

        ShowErrorPanel();
    }
    catch (Exception ex)
    {
        MessageBox.Show(
            ex.ToString(),
            "Parse Your SQL",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
    }
}
