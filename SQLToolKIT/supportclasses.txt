using Microsoft.SqlServer.TransactSql.ScriptDom;
using ScriptFormatter.Core.Review.ScriptFormatter.Core.Review;
using System;
using System.Collections.Generic;
using System.Linq;

namespace ScriptFormatter.Core.Review.Rules
{
    public sealed class LeftJoinFilterPredicateRule
        : ReviewRuleBase
    {
        public override string RuleId
        {
            get { return "SQLR0023"; }
        }

        public override ReviewSeverity Severity
        {
            get { return ReviewSeverity.Information; }
        }

        public override IEnumerable<ReviewFinding> Review(
            SqlReviewContext context)
        {
            var findings =
                new List<ReviewFinding>();

            if (context == null ||
                context.Fragment == null)
            {
                return findings;
            }

            var visitor =
                new Visitor(this);

            context.Fragment.Accept(
                visitor);

            return visitor.Findings;
        }

        private sealed class Visitor
            : TSqlFragmentVisitor
        {
            private readonly LeftJoinFilterPredicateRule _rule;

            public Visitor(
                LeftJoinFilterPredicateRule rule)
            {
                _rule = rule;

                Findings =
                    new List<ReviewFinding>();
            }

            public IList<ReviewFinding> Findings
            {
                get;
                private set;
            }

            public override void Visit(
                QuerySpecification node)
            {
                if (node == null)
                {
                    return;
                }

                IList<QualifiedJoin> leftJoins =
                    GetLeftJoinsFromCurrentQuery(
                        node);

                if (leftJoins.Count == 0)
                {
                    return;
                }

                //
                // PART 1
                //
                // Review filter predicates inside
                // LEFT JOIN ON conditions.
                //
                foreach (QualifiedJoin join
                    in leftJoins)
                {
                    ReviewJoinOnClause(
                        join);
                }

                //
                // PART 2
                //
                // Review WHERE predicates referencing
                // right-side LEFT JOIN aliases.
                //
                if (node.WhereClause == null ||
                    node.WhereClause.SearchCondition == null)
                {
                    return;
                }

                IList<string> rightAliases =
                    leftJoins
                        .Select(GetRightSideAlias)
                        .Where(
                            x =>
                                !string.IsNullOrWhiteSpace(
                                    x))
                        .Distinct(
                            StringComparer.OrdinalIgnoreCase)
                        .ToList();

                if (rightAliases.Count == 0)
                {
                    return;
                }

                var whereVisitor =
                    new WherePredicateVisitor(
                        rightAliases);

                node.WhereClause
                    .SearchCondition
                    .Accept(
                        whereVisitor);

                foreach (TSqlFragment predicate
                    in whereVisitor.Predicates)
                {
                    Findings.Add(
                        _rule.CreateFinding(
                            predicate,
                            "WHERE condition references a LEFT JOIN table and may change LEFT JOIN behavior. Review whether the condition belongs in ON or whether INNER JOIN was intended."));
                }
            }

            private void ReviewJoinOnClause(
                QualifiedJoin join)
            {
                if (join == null ||
                    join.SearchCondition == null)
                {
                    return;
                }

                var predicateVisitor =
                    new OnFilterPredicateVisitor();

                join.SearchCondition.Accept(
                    predicateVisitor);

                foreach (BooleanComparisonExpression predicate
                    in predicateVisitor.Predicates)
                {
                    Findings.Add(
                        _rule.CreateFinding(
                            predicate,
                            "LEFT JOIN ON contains a filter predicate. Review whether the condition belongs in ON or WHERE."));
                }
            }

            private static IList<QualifiedJoin>
                GetLeftJoinsFromCurrentQuery(
                    QuerySpecification query)
            {
                var joins =
                    new List<QualifiedJoin>();

                if (query == null ||
                    query.FromClause == null ||
                    query.FromClause.TableReferences == null)
                {
                    return joins;
                }

                foreach (TableReference tableReference
                    in query.FromClause.TableReferences)
                {
                    CollectLeftJoins(
                        tableReference,
                        joins);
                }

                return joins;
            }

            private static void CollectLeftJoins(
                TableReference tableReference,
                IList<QualifiedJoin> joins)
            {
                if (tableReference == null)
                {
                    return;
                }

                var qualifiedJoin =
                    tableReference as QualifiedJoin;

                if (qualifiedJoin != null)
                {
                    if (qualifiedJoin.QualifiedJoinType ==
                        QualifiedJoinType.LeftOuter)
                    {
                        joins.Add(
                            qualifiedJoin);
                    }

                    //
                    // Walk only the JOIN tree belonging
                    // to the current query scope.
                    //
                    CollectLeftJoins(
                        qualifiedJoin.FirstTableReference,
                        joins);

                    CollectLeftJoins(
                        qualifiedJoin.SecondTableReference,
                        joins);

                    return;
                }

                var joinParenthesis =
                    tableReference
                        as JoinParenthesisTableReference;

                if (joinParenthesis != null)
                {
                    CollectLeftJoins(
                        joinParenthesis.Join,
                        joins);

                    return;
                }

                //
                // Do not enter QueryDerivedTable or
                // other nested SELECTs here.
                //
                // Nested QuerySpecification nodes will
                // be reviewed independently.
                //
            }

            private static string GetRightSideAlias(
                QualifiedJoin join)
            {
                if (join == null ||
                    join.SecondTableReference == null)
                {
                    return null;
                }

                var tableWithAlias =
                    join.SecondTableReference
                        as TableReferenceWithAlias;

                if (tableWithAlias == null ||
                    tableWithAlias.Alias == null)
                {
                    return null;
                }

                return tableWithAlias.Alias.Value;
            }
        }

        private sealed class OnFilterPredicateVisitor
            : TSqlFragmentVisitor
        {
            public OnFilterPredicateVisitor()
            {
                Predicates =
                    new List<BooleanComparisonExpression>();
            }

            public IList<BooleanComparisonExpression> Predicates
            {
                get;
                private set;
            }

            public override void Visit(
                BooleanComparisonExpression node)
            {
                if (node == null)
                {
                    return;
                }

                bool firstIsColumn =
                    node.FirstExpression
                        is ColumnReferenceExpression;

                bool secondIsColumn =
                    node.SecondExpression
                        is ColumnReferenceExpression;

                //
                // Normal JOIN relationship:
                //
                // A.CustomerId = B.CustomerId
                //
                if (firstIsColumn &&
                    secondIsColumn)
                {
                    return;
                }

                //
                // Review examples:
                //
                // B.Status = 'A'
                // B.Type = @V_Type
                // B.Date = GETDATE()
                //
                if (firstIsColumn ||
                    secondIsColumn)
                {
                    Predicates.Add(
                        node);
                }
            }

            //
            // Do not inspect predicates inside
            // nested scalar subqueries.
            //
            public override void ExplicitVisit(
                ScalarSubquery node)
            {
                return;
            }
        }

        private sealed class WherePredicateVisitor
            : TSqlFragmentVisitor
        {
            private readonly IList<string> _rightAliases;

            public WherePredicateVisitor(
                IList<string> rightAliases)
            {
                _rightAliases =
                    rightAliases;

                Predicates =
                    new List<TSqlFragment>();
            }

            public IList<TSqlFragment> Predicates
            {
                get;
                private set;
            }

            public override void Visit(
                BooleanComparisonExpression node)
            {
                if (node == null)
                {
                    return;
                }

                if (ReferencesRightAlias(
                        node))
                {
                    Predicates.Add(
                        node);
                }
            }

            public override void Visit(
                BooleanIsNullExpression node)
            {
                if (node == null)
                {
                    return;
                }

                //
                // WHERE B.Id IS NULL
                //
                // Usually intentional anti-join logic,
                // so do not flag.
                //
                if (!node.IsNot)
                {
                    return;
                }

                //
                // WHERE B.Id IS NOT NULL
                //
                // Removes unmatched LEFT JOIN rows.
                //
                if (ReferencesRightAlias(
                        node))
                {
                    Predicates.Add(
                        node);
                }
            }

            //
            // Critical:
            //
            // Do not inspect nested SELECTs as part
            // of the current WHERE predicate.
            //
            public override void ExplicitVisit(
                ScalarSubquery node)
            {
                return;
            }

            private bool ReferencesRightAlias(
                TSqlFragment fragment)
            {
                var visitor =
                    new ColumnAliasVisitor(
                        _rightAliases);

                fragment.Accept(
                    visitor);

                return visitor.Found;
            }
        }

        private sealed class ColumnAliasVisitor
            : TSqlFragmentVisitor
        {
            private readonly IList<string> _aliases;

            public ColumnAliasVisitor(
                IList<string> aliases)
            {
                _aliases =
                    aliases;
            }

            public bool Found
            {
                get;
                private set;
            }

            public override void Visit(
                ColumnReferenceExpression node)
            {
                if (node == null ||
                    node.MultiPartIdentifier == null ||
                    node.MultiPartIdentifier
                        .Identifiers.Count < 2)
                {
                    return;
                }

                int identifierCount =
                    node.MultiPartIdentifier
                        .Identifiers.Count;

                string alias =
                    node.MultiPartIdentifier
                        .Identifiers[
                            identifierCount - 2]
                        .Value;

                if (_aliases.Any(
                    x =>
                        string.Equals(
                            x,
                            alias,
                            StringComparison.OrdinalIgnoreCase)))
                {
                    Found =
                        true;
                }
            }

            //
            // Do not accidentally match aliases
            // inside a nested query.
            //
            public override void ExplicitVisit(
                ScalarSubquery node)
            {
                return;
            }
        }
    }
}
