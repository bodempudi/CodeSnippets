using Microsoft.SqlServer.TransactSql.ScriptDom;
using System.Collections.Generic;

namespace NBK.SQL.CodingStandards.Core.Models
{
    public sealed class SqlParseResult
    {
        public TSqlFragment Fragment
        {
            get;
            set;
        }

        public IList<SqlResultInfo> Errors
        {
            get;
            set;
        }

        public bool IsValid
        {
            get
            {
                return Errors == null ||
                       Errors.Count == 0;
            }
        }
    }
}
