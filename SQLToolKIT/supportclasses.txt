using Microsoft.SqlServer.TransactSql.ScriptDom;
using ScriptFormatter.Core.Review.ScriptFormatter.Core.Review;
using System.Collections.Generic;

namespace ScriptFormatter.Core.Review.Rules
{
    public sealed class FunctionOnJoinColumnRule
        : ReviewRuleBase
    {
        public override string RuleId
        {
            get { return "SQLR0025"; }
        }

        public override ReviewSeverity Severity
        {
            get { return ReviewSeverity.Warning; }
        }

        public override IEnumerable<ReviewFinding> Review(
            SqlReviewContext context)
        {
            var findings =
                new List<ReviewFinding>();

            if (context == null ||
                context.Fragment == null)
            {
                return findings;
            }

            var visitor =
                new Visitor(
                    this);

            context.Fragment.Accept(
                visitor);

            return visitor.Findings;
        }

        private sealed class Visitor
            : TSqlFragmentVisitor
        {
            private readonly FunctionOnJoinColumnRule _rule;

            public Visitor(
                FunctionOnJoinColumnRule rule)
            {
                _rule =
                    rule;

                Findings =
                    new List<ReviewFinding>();
            }

            public IList<ReviewFinding> Findings
            {
                get;
                private set;
            }

            public override void ExplicitVisit(
                QualifiedJoin node)
            {
                if (node == null)
                {
                    return;
                }

                if (node.SearchCondition != null)
                {
                    var predicateVisitor =
                        new JoinPredicateVisitor(
                            _rule,
                            Findings);

                    node.SearchCondition.Accept(
                        predicateVisitor);
                }

                //
                // Continue normal traversal.
                //
                // Nested queries and nested JOINs will therefore
                // be reviewed independently.
                //
                base.ExplicitVisit(
                    node);
            }
        }

        private sealed class JoinPredicateVisitor
            : TSqlFragmentVisitor
        {
            private readonly FunctionOnJoinColumnRule _rule;

            private readonly IList<ReviewFinding> _findings;

            public JoinPredicateVisitor(
                FunctionOnJoinColumnRule rule,
                IList<ReviewFinding> findings)
            {
                _rule =
                    rule;

                _findings =
                    findings;
            }

            //
            // Examples:
            //
            // ON UPPER(A.Code) = B.Code
            //
            // ON ISNULL(A.CustomerId, 0) = B.CustomerId
            //
            // ON YEAR(A.CreatedDate) = B.CreatedYear
            //
            public override void Visit(
                FunctionCall node)
            {
                if (node == null)
                {
                    return;
                }

                if (!ContainsColumn(
                        node))
                {
                    return;
                }

                AddFinding(
                    node);
            }

            //
            // Example:
            //
            // ON CONVERT(DATE, A.CreatedDate) =
            //    B.BusinessDate
            //
            public override void Visit(
                ConvertCall node)
            {
                if (node == null ||
                    node.Parameter == null)
                {
                    return;
                }

                if (!ContainsColumn(
                        node.Parameter))
                {
                    return;
                }

                AddFinding(
                    node);
            }

            //
            // Example:
            //
            // ON CAST(A.CreatedDate AS DATE) =
            //    B.BusinessDate
            //
            public override void Visit(
                CastCall node)
            {
                if (node == null ||
                    node.Parameter == null)
                {
                    return;
                }

                if (!ContainsColumn(
                        node.Parameter))
                {
                    return;
                }

                AddFinding(
                    node);
            }

            //
            // Important:
            //
            // Do not inspect a nested SELECT as part of
            // the parent JOIN condition.
            //
            // Example:
            //
            // ON A.BusinessDate =
            // (
            //     SELECT MAX(BusinessDate)
            //     FROM dbo.TableB
            // )
            //
            // MAX(BusinessDate) is not a function applied
            // to a column in the parent JOIN predicate.
            //
            public override void ExplicitVisit(
                ScalarSubquery node)
            {
                return;
            }

            private void AddFinding(
                TSqlFragment fragment)
            {
                _findings.Add(
                    _rule.CreateFinding(
                        fragment,
                        "Function applied to a column in JOIN condition."));
            }

            private static bool ContainsColumn(
                TSqlFragment fragment)
            {
                if (fragment == null)
                {
                    return false;
                }

                var columnVisitor =
                    new ColumnVisitor();

                fragment.Accept(
                    columnVisitor);

                return columnVisitor.HasColumn;
            }

            private sealed class ColumnVisitor
                : TSqlFragmentVisitor
            {
                public bool HasColumn
                {
                    get;
                    private set;
                }

                public override void Visit(
                    ColumnReferenceExpression node)
                {
                    HasColumn =
                        true;
                }

                //
                // Don't consider columns belonging to
                // a nested scalar subquery.
                //
                public override void ExplicitVisit(
                    ScalarSubquery node)
                {
                    return;
                }
            }
        }
    }
}
