using Microsoft.SqlServer.TransactSql.ScriptDom;
using System.Collections.Generic;

namespace ScriptFormatter.Core.Review.Rules
{
    public sealed class UseStatementRule
        : ReviewRuleBase
    {
        public override string RuleId
        {
            get { return "SQLR0025"; }
        }

        public override ReviewSeverity Severity
        {
            get { return ReviewSeverity.Warning; }
        }

        public override IEnumerable<ReviewFinding> Review(
            SqlReviewContext context)
        {
            var visitor =
                new Visitor();

            context.Fragment.Accept(
                visitor);

            var findings =
                new List<ReviewFinding>();

            if (visitor.UseStatements.Count == 0)
            {
                findings.Add(
                    CreateFinding(
                        context.Fragment,
                        "USE statement is missing. " +
                        "A SQL file must define one database context."));

                return findings;
            }

            if (visitor.UseStatements.Count > 1)
            {
                UseStatement secondUse =
                    visitor.UseStatements[1];

                findings.Add(
                    CreateFinding(
                        secondUse,
                        "Multiple USE statements detected. " +
                        "A SQL file must define only one database context."));
            }

            return findings;
        }

        private sealed class Visitor
            : TSqlFragmentVisitor
        {
            public Visitor()
            {
                UseStatements =
                    new List<UseStatement>();
            }

            public IList<UseStatement> UseStatements
            {
                get;
                private set;
            }

            public override void Visit(
                UseStatement node)
            {
                if (node == null)
                {
                    return;
                }

                UseStatements.Add(
                    node);
            }
        }
    }
}
