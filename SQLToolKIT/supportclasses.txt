using Microsoft.SqlServer.TransactSql.ScriptDom;
using System.Collections.Generic;

namespace NBK.SQL.CodingStandards.Core.Models
{
    public sealed class SqlParseResult
    {
        public TSqlFragment Fragment
        {
            get;
            set;
        }

        public IList<SqlResultInfo> Errors
        {
            get;
            set;
        }

        public bool IsValid
        {
            get
            {
                return Errors == null ||
                       Errors.Count == 0;
            }
        }

        public bool CanReview
        {
            get
            {
                return Fragment != null;
            }
        }
    }
}

private static void ReviewYourSQL()
{
    try
    {
        string sql =
            GetCurrentEditorText();

        var parserService =
            new SQLParserService();

        SqlParseResult parseResult =
            parserService.ParseSql(
                sql);

        var reviewResults =
            new List<SqlResultInfo>();

        //
        // Keep parse errors.
        //
        if (!parseResult.IsValid)
        {
            if (parseResult.Errors != null)
            {
                reviewResults.AddRange(
                    parseResult.Errors);
            }

            //
            // No fragment means review cannot safely continue.
            //
            if (!parseResult.CanReview)
            {
                EnsureErrorPanel();

                _errorPanel.SetTitle(
                    "SQL Review Results");

                _errorPanel.ShowResults(
                    reviewResults);

                ShowErrorPanel();

                return;
            }

            //
            // Fragment exists, so review can continue,
            // but tell the developer that results may
            // be incomplete.
            //
            reviewResults.Add(
                new SqlResultInfo
                {
                    Source = "Review",
                    Severity = "Information",
                    Line = 0,
                    Column = 0,
                    Message =
                        "SQL parsing errors were detected. Review results may be incomplete."
                });
        }

        var reviewService =
            new SqlReviewService();

        IEnumerable<ReviewFinding> findings =
            reviewService.Review(
                sql,
                parseResult);

        if (findings != null)
        {
            foreach (ReviewFinding finding
                in findings)
            {
                reviewResults.Add(
                    new SqlResultInfo
                    {
                        Source =
                            finding.RuleId,

                        Severity =
                            finding.Severity.ToString(),

                        Line =
                            finding.Line,

                        Column =
                            finding.Column,

                        Message =
                            finding.Message,

                        StartOffset =
                            finding.StartOffset,

                        Length =
                            finding.Length
                    });
            }
        }

        //
        // Only show "no violations" when:
        // 1. parsing was valid, and
        // 2. no review findings were produced.
        //
        if (parseResult.IsValid &&
            reviewResults.Count == 0)
        {
            reviewResults.Add(
                new SqlResultInfo
                {
                    Source = "Review",
                    Severity = "Information",
                    Line = 0,
                    Column = 0,
                    Message =
                        "No review violations found."
                });
        }

        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Review Results");

        _errorPanel.ShowResults(
            reviewResults);

        ShowErrorPanel();
    }
    catch (Exception ex)
    {
        MessageBox.Show(
            ex.ToString(),
            "Review Your SQL",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
    }
}
