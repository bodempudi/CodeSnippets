using Microsoft.SqlServer.TransactSql.ScriptDom;
using ScriptFormatter.Core.Review.ScriptFormatter.Core.Review;
using System;
using System.Collections.Generic;
using System.Linq;

namespace ScriptFormatter.Core.Review.Rules
{
    public sealed class NoLockRule
        : ReviewRuleBase
    {
        public override string RuleId
        {
            get { return "SQLR0001"; }
        }

        public override ReviewSeverity Severity
        {
            get { return ReviewSeverity.Warning; }
        }

        public override IEnumerable<ReviewFinding> Review(
            SqlReviewContext context)
        {
            var findings =
                new List<ReviewFinding>();

            if (context == null ||
                context.Fragment == null)
            {
                return findings;
            }

            //
            // Collect CTE names first so that references
            // such as FROM CustomerCTE are not treated
            // as physical tables requiring NOLOCK.
            //
            var cteVisitor =
                new CteVisitor();

            context.Fragment.Accept(
                cteVisitor);

            var visitor =
                new Visitor(
                    this,
                    cteVisitor.CteNames);

            context.Fragment.Accept(
                visitor);

            return visitor.Findings;
        }

        private sealed class Visitor
            : TSqlFragmentVisitor
        {
            private readonly NoLockRule _rule;

            private readonly ISet<string> _cteNames;

            public Visitor(
                NoLockRule rule,
                ISet<string> cteNames)
            {
                _rule =
                    rule;

                _cteNames =
                    cteNames;

                Findings =
                    new List<ReviewFinding>();
            }

            public IList<ReviewFinding> Findings
            {
                get;
                private set;
            }

            public override void Visit(
                QuerySpecification node)
            {
                if (node == null ||
                    node.FromClause == null ||
                    node.FromClause.TableReferences == null)
                {
                    return;
                }

                //
                // Only inspect tables appearing in FROM/JOIN.
                //
                // This avoids treating INSERT, UPDATE or DELETE
                // target objects as read tables.
                //
                foreach (TableReference tableReference
                    in node.FromClause.TableReferences)
                {
                    ReviewTableReference(
                        tableReference);
                }
            }

            private void ReviewTableReference(
                TableReference tableReference)
            {
                if (tableReference == null)
                {
                    return;
                }

                var namedTable =
                    tableReference
                        as NamedTableReference;

                if (namedTable != null)
                {
                    ReviewNamedTable(
                        namedTable);

                    return;
                }

                var qualifiedJoin =
                    tableReference
                        as QualifiedJoin;

                if (qualifiedJoin != null)
                {
                    ReviewTableReference(
                        qualifiedJoin.FirstTableReference);

                    ReviewTableReference(
                        qualifiedJoin.SecondTableReference);

                    return;
                }

                var unqualifiedJoin =
                    tableReference
                        as UnqualifiedJoin;

                if (unqualifiedJoin != null)
                {
                    ReviewTableReference(
                        unqualifiedJoin.FirstTableReference);

                    ReviewTableReference(
                        unqualifiedJoin.SecondTableReference);

                    return;
                }

                var joinParenthesis =
                    tableReference
                        as JoinParenthesisTableReference;

                if (joinParenthesis != null)
                {
                    ReviewTableReference(
                        joinParenthesis.Join);

                    return;
                }

                //
                // QueryDerivedTable is intentionally not
                // traversed here.
                //
                // Its QuerySpecification will be visited
                // independently by the main visitor.
                //
            }

            private void ReviewNamedTable(
                NamedTableReference table)
            {
                if (table == null ||
                    table.SchemaObject == null)
                {
                    return;
                }

                if (IsSystemObject(
                        table))
                {
                    return;
                }

                if (IsTemporaryTable(
                        table))
                {
                    return;
                }

                if (IsCteReference(
                        table))
                {
                    return;
                }

                if (HasNoLock(
                        table))
                {
                    return;
                }

                Findings.Add(
                    _rule.CreateFinding(
                        table,
                        "NOLOCK missing: " +
                        GetTableName(table)));
            }

            private bool IsCteReference(
                NamedTableReference table)
            {
                IList<Identifier> identifiers =
                    table.SchemaObject.Identifiers;

                if (identifiers == null ||
                    identifiers.Count != 1)
                {
                    return false;
                }

                string objectName =
                    identifiers[0].Value;

                return _cteNames.Contains(
                    objectName);
            }

            private static bool IsTemporaryTable(
                NamedTableReference table)
            {
                IList<Identifier> identifiers =
                    table.SchemaObject.Identifiers;

                if (identifiers == null ||
                    identifiers.Count == 0)
                {
                    return false;
                }

                string objectName =
                    identifiers[
                        identifiers.Count - 1]
                        .Value;

                return !string.IsNullOrWhiteSpace(
                           objectName) &&
                       objectName.StartsWith(
                           "#",
                           StringComparison.Ordinal);
            }

            private static bool IsSystemObject(
                NamedTableReference table)
            {
                IList<Identifier> identifiers =
                    table.SchemaObject.Identifiers;

                if (identifiers == null ||
                    identifiers.Count < 2)
                {
                    return false;
                }

                //
                // Examples:
                //
                // sys.tables
                //
                // tempdb.sys.tables
                //
                // master.sys.objects
                //
                // INFORMATION_SCHEMA.TABLES
                //
                string schemaName =
                    identifiers[
                        identifiers.Count - 2]
                        .Value;

                if (string.Equals(
                        schemaName,
                        "sys",
                        StringComparison.OrdinalIgnoreCase))
                {
                    return true;
                }

                if (string.Equals(
                        schemaName,
                        "INFORMATION_SCHEMA",
                        StringComparison.OrdinalIgnoreCase))
                {
                    return true;
                }

                return false;
            }

            private static bool HasNoLock(
                NamedTableReference table)
            {
                if (table.TableHints == null ||
                    table.TableHints.Count == 0)
                {
                    return false;
                }

                return table.TableHints.Any(
                    hint =>
                        hint != null &&
                        hint.HintKind ==
                            TableHintKind.NoLock);
            }

            private static string GetTableName(
                NamedTableReference table)
            {
                if (table == null ||
                    table.SchemaObject == null ||
                    table.SchemaObject.Identifiers == null)
                {
                    return string.Empty;
                }

                return string.Join(
                    ".",
                    table.SchemaObject
                        .Identifiers
                        .Select(x => x.Value));
            }
        }

        private sealed class CteVisitor
            : TSqlFragmentVisitor
        {
            public CteVisitor()
            {
                CteNames =
                    new HashSet<string>(
                        StringComparer.OrdinalIgnoreCase);
            }

            public ISet<string> CteNames
            {
                get;
                private set;
            }

            public override void Visit(
                CommonTableExpression node)
            {
                if (node == null ||
                    node.ExpressionName == null ||
                    string.IsNullOrWhiteSpace(
                        node.ExpressionName.Value))
                {
                    return;
                }

                CteNames.Add(
                    node.ExpressionName.Value);
            }
        }
    }
}
