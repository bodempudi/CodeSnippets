private static void ReviewSQL()
{
    try
    {
        string sql =
            GetCurrentEditorText();

        var reviewService =
            new SqlReviewService();

        IList<ParseError> parseErrors;

        IList<ReviewFinding> findings =
            reviewService.Review(
                sql,
                out parseErrors);

        //
        // Parser errors found while reviewing
        //
        if (parseErrors != null &&
            parseErrors.Count > 0)
        {
            var parseResults =
                new List<SqlResultInfo>();

            foreach (
                ParseError error
                in parseErrors)
            {
                parseResults.Add(
                    new SqlResultInfo
                    {
                        Source =
                            "Parser",

                        Severity =
                            "Error",

                        Line =
                            error.Line,

                        Column =
                            error.Column,

                        Message =
                            error.Message
                    });
            }



            EnsureErrorPanel();

            _errorPanel.SetTitle(
                "SQL Parsing Results");

            _errorPanel.ShowResults(
                parseResults);

            ShowErrorPanel();

            return;
        }

        //
        // Review findings
        //
        var reviewResults =
            new List<SqlResultInfo>();

        if (findings != null)
        {
            foreach (
                ReviewFinding finding
                in findings)
            {
                reviewResults.Add(
                    new SqlResultInfo
                    {
                        Source =
                            finding.RuleId,

                        Severity =
                            finding.Severity.ToString(),

                        Line =
                            finding.Line,

                        Column =
                            finding.Column,

                        Message =
                            finding.Message,

                        StartOffset =
                            finding.StartOffset,

                        Length =
                            finding.Length
                    });
            }
        }
        if (reviewResults.Count == 0)
        {
            reviewResults.Add(
                new SqlResultInfo
                {
                    Source = "Review",
                    Severity = "Info",
                    Line = 0,
                    Column = 0,
                    Message = "No review violations found."
                });
        }
        EnsureErrorPanel();

        _errorPanel.SetTitle(
            "SQL Review Results");

        _errorPanel.ShowResults(
            reviewResults);

        ShowErrorPanel();
    }
    catch (Exception ex)
    {
        MessageBox.Show(
            ex.ToString(),
            "Review Your SQL",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
    }
}
