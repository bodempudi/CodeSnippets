private static IList<SqlErrorInfo>
	ConvertValidationIssues(
		IList<SqlValidationIssue> issues)
{
	var result =
		new List<SqlErrorInfo>();

	if (issues == null)
	{
		return result;
	}

	foreach (SqlValidationIssue issue in issues)
	{
		result.Add(
			new SqlErrorInfo
			{
				Line = issue.Line,
				Column = issue.Column,
				Message = issue.Message,
				Source = "Validator"
			});
	}

	return result;
}
