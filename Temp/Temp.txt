private static void FormatCurrentSql()
{
	try
	{
		string sql =
			GetCurrentEditorText();

		IList<ParseError> parseErrors =
			ParseSql(sql);

		if (parseErrors != null &&
			parseErrors.Count > 0)
		{
			ShowSqlErrors(
				ConvertParseErrors(parseErrors));

			return;
		}

		var validator =
			new SqlValidatorService();

		IList<SqlValidationIssue>
			validationIssues =
				validator.Validate(sql);

		if (validationIssues != null &&
			validationIssues.Count > 0)
		{
			ShowSqlErrors(
				ConvertValidationIssues(
					validationIssues));

			return;
		}

		ShowSqlErrors(null);
	}
	catch (Exception ex)
	{
		MessageBox.Show(
			ex.ToString(),
			"Venkat SQL Formatter",
			MessageBoxButtons.OK,
			MessageBoxIcon.Error);
	}
}
