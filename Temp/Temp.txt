private static IList<SqlErrorInfo> ConvertParseErrors(
	IList<ParseError> errors)
{
	var result = new List<SqlErrorInfo>();

	if (errors == null)
	{
		return result;
	}

	foreach (ParseError error in errors)
	{
		result.Add(
			new SqlErrorInfo
			{
				Line = error.Line,
				Column = error.Column,
				Message = error.Message,
				Source = "Parser"
			});
	}

	return result;
}
