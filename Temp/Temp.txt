using System.Collections.Generic;
using System.IO;
using Microsoft.SqlServer.TransactSql.ScriptDom;

namespace ScriptFormatter.Core.Formatting.Custom
{
	public sealed class CustomSqlFormatterService
	{
		public string Format(string sql)
		{
			var parser =
				new TSql160Parser(false);

			TSqlFragment fragment;

			IList<ParseError> errors;

			using (var reader =
				new StringReader(sql))
			{
				fragment =
					parser.Parse(
						reader,
						out errors);
			}

			if (errors != null &&
				errors.Count > 0)
			{
				return sql;
			}

			// For now fallback to ScriptDOM generator.
			// Next step: detect QuerySpecification and custom format.
			var generator =
				new Sql160ScriptGenerator();

			string output;

			generator.GenerateScript(
				fragment,
				out output);

			return output;
		}
	}
}
