private static void GoToEditorPosition(
	int line,
	int column)
{
	IntPtr sci =
		PluginBase.GetCurrentScintilla();

	int zeroLine =
		line - 1;

	int zeroColumn =
		column - 1;

	Win32.SendMessage(
		sci,
		(uint)SciMsg.SCI_GOTOLINE,
		zeroLine,
		0);

	int position =
		(int)Win32.SendMessage(
			sci,
			(uint)SciMsg.SCI_FINDCOLUMN,
			zeroLine,
			zeroColumn);

	Win32.SendMessage(
		sci,
		(uint)SciMsg.SCI_GOTOPOS,
		position,
		0);
}
