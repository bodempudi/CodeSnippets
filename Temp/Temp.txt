using Microsoft.SqlServer.TransactSql.ScriptDom;

namespace ScriptFormatter.Core.Formatting.Custom.Visitors
{
	public sealed class QuerySpecificationVisitor
		: TSqlFragmentVisitor
	{
		public QuerySpecification QuerySpecification
		{
			get;
			private set;
		}

		public override void ExplicitVisit(
			QuerySpecification node)
		{
			// Capture first SELECT query block.
			if (QuerySpecification == null)
			{
				QuerySpecification = node;
			}
		}
	}
}
